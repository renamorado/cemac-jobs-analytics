\section{BDF Cleaning Results}

\begin{frame}{Data cleaning overview}
    \begin{itemize}
        \item The working panel is organized at the firm-year level, so duplicate handling changes both sample size and identification.
        \item Source sectors are three-digit NACAM Rev.1 branches. The reviewed crosswalk assigns them to 16 categories based on the NACAM--ISIC Rev.4 crosswalk.
        \item Duplicate treatment and the documented section assignments determine the analysis sample and sector interpretation.
    \end{itemize}
\end{frame}

\begin{frame}{Firm-Year Duplicate Cleaning}
    \footnotesize
    \begin{itemize}
        \item Working key: \texttt{firmid} $\times$ \texttt{fin\_yr}
        \item Perfect duplicates would be collapsed with original row order as the tie-breaker.
        \item Conflicting duplicate firm-years are fully removed from the cleaned analysis file, while we figure out the best approach for handling them.
    \end{itemize}

\begin{frame}{Cleaning Takeaways}
    \begin{itemize}
        \item The cleaned panel resizes the sample from 9,325 to 9,068 observations after removing duplicate firm-years.
        \item There are 127 conflicting firm-years, accounting for 257 dropped obs. Those observations are excluded rather than resolved ad hoc.
        \item Ecofin uses ISIC Rev.4 sections based on the NACAM--ISIC Rev.4 crosswalk. The combinations are professional \& admin. support, and arts/recreation, other services \& household employers (division 97). Crosswalk qualifications are listed in the appendix.
    \end{itemize}

\section{Cameroun Sectoral Elasticities}

\begin{frame}{Specification and Sample}
    \footnotesize
    \begin{equation}
        \ln(E_{it}) = \alpha_i + \delta_{st} + \sum_s \beta_s \left( \ln(X_{it}) \times D_{ist} \right) + \varepsilon_{it}
    \end{equation}

    \begin{itemize}
        \item The unit of observation is a firm-year $(i,t)$.
        \item $E_{it}$ is employment for firm $i$ in year $t$.
        \item $X_{it}$ is either value added or total revenue, estimated in separate regressions.
        \item $D_{ist}$ equals 1 if firm $i$ belongs to ISIC section $s$ in year $t$, and 0 otherwise.
        \item Group membership follows the reported classification in each firm-year; firms may change groups over time.
        \item $\alpha_i$ are firm fixed effects; $\delta_{st}$ are section--by-fiscal-year fixed effects.
        \item $\beta_s$ is the elasticity for firms in sector $s$, identified through the interaction between $\ln(X_{it})$ and the sector indicator $D_{ist}$.
        \item No additional firm-level controls are included beyond these fixed effects.
        \item Standard errors are clustered at the firm level; reported sectors have at least 30 observations and 10 firms in both outcome samples.
    \end{itemize}
\end{frame}

\begin{frame}{Employment Distribution by Fiscal Year}
    \centering
    \includegraphics[width=\textwidth,height=0.70\textheight,keepaspectratio]{../output/figures/cmr_isic4_section_results_en_labels_ln_emp_density_by_year.pdf}
    \bdffigsource
\end{frame}

\begin{frame}{Value-Added Elasticity Specification}
    \small
    \begin{equation}
        \ln(E_{it}) = \alpha_i + \delta_{st} + \sum_s \beta^{VA}_s \left( \ln(VA_{it}) \times D_{ist} \right) + \varepsilon_{it}
    \end{equation}

    \begin{itemize}
        \item Estimates use the Cameroon administrative firm-year panel.
        \item Firm fixed effects absorb time-invariant firm differences at the taxpayer/firm level.
        \item Section--by-fiscal-year fixed effects absorb annual shocks common to firms in the same section.
        \item No age, ownership, exporter-status, or other firm controls are included.
        \item The plotted point for each sector is $\beta^{VA}_s$ with a firm-clustered 95 percent interval.
    \end{itemize}
\end{frame}

\begin{frame}{Total-Revenue Elasticity Specification}
    \small
    \begin{equation}
        \ln(E_{it}) = \alpha_i + \delta_{st} + \sum_s \beta^{R}_s \left( \ln(R_{it}) \times D_{ist} \right) + \varepsilon_{it}
    \end{equation}

    \begin{itemize}
        \item $R_{it}$ is total revenue, estimated in a separate regression from value added.
        \item Fixed effects are at the firm level and section--by-fiscal-year level.
        \item No additional firm-level controls are included beyond these fixed effects.
        \item The plotted point for each sector is $\beta^{R}_s$ with standard errors clustered at the firm level.
    \end{itemize}
\end{frame}

\begin{frame}{Fixed-Effect Robustness Check}
    \begin{itemize}

        \item Baseline: firm FE plus detailed section--by-fiscal-year FE.
        \item Looser alternative: firm FE plus fiscal-year FE only.
        \item Intermediate alternative: firm FE plus broad-activity-group-by-fiscal-year FE.
        \item No age, ownership, exporter-status, or other firm-level controls are added in these robustness checks.
    \end{itemize}

\begin{frame}{Census Turnover-Employment Specification}
    \footnotesize
    \begin{equation}
        \ln(E_i) = \alpha_s + \sum_s \beta_s
        \left[\ln(R_i) \times D_{is}\right] + \varepsilon_i
    \end{equation}

    \begin{itemize}
        \item The unit of observation is a Census headquarters firm $i$.
        \item $E_i$ is employment and $R_i$ is annual turnover from RGE3.
        \item $D_{is}$ equals 1 if firm $i$ belongs to ISIC Rev.4 division $s$.
        \item $\alpha_s$ are detailed ISIC Rev.4 division intercepts.
        \item $\beta_s$ is the cross-sectional employment elasticity with respect to annual turnover in sector $s$.
        \item There are no firm fixed effects, year fixed effects, or division--year fixed effects because the Census has no panel time variation.
        \item Standard errors are heteroskedasticity-robust; reported sectors have at least 30 positive-employment and positive-turnover headquarters firms.
    \end{itemize}
\end{frame}

\begin{frame}{Census Exporter-Interaction Specification}
    \footnotesize
    \begin{equation}
        \ln(E_i) = \alpha_s + \beta_s \ln(R_i) + \gamma_s Exporter_i
        + \delta_s \left[\ln(R_i) \times Exporter_i\right] + \varepsilon_i,
        \quad i \in s
    \end{equation}

    \begin{itemize}
        \item The unit of observation is a Census headquarters firm $i$ in ISIC Rev.4 division $s$.
        \item $E_i$ is employment and $R_i$ is annual turnover from RGE3.
        \item $Exporter_i$ equals 1 when embedded Census export turnover is positive.
        \item $\beta_s$ is the non-exporter elasticity; $\beta_s+\delta_s$ is the exporter elasticity.
        \item The total elasticity is estimated separately among all firms in the same reported sector sample.
        \item Standard errors are heteroskedasticity-robust.
        \item Reported sectors have at least 30 usable firms, 10 exporters, and 10 non-exporters.
    \end{itemize}
\end{frame}

\section{B-READY Mapped WBES Constraints}

\begin{frame}[shrink=10]{B-READY/WBES Constraint Extension}
    \begin{itemize}

\begin{frame}{Focal WBES Revenue-Exporter Specification}
    \small
    \begin{equation}
        \ln(E_i) = \alpha_c + \beta_c \ln(R_i) + \gamma_c Exporter_i
        + \delta_c\left[\ln(R_i) \times Exporter_i\right]
        + \theta_c Z_i + \lambda_{c q(i)} + \varepsilon_i,
        \quad i \in c
    \end{equation}

    \begin{itemize}
        \item $c$ indexes the country. The model is estimated separately within each country.
        \item WBES is cross-sectional, so there are no firm fixed effects or panel controls.
        \item $Z_i$ controls for log firm age, foreign ownership share, and government/state ownership share.
        \item $\lambda_{c q(i)}$ are country-specific ISIC Rev.4 section fixed effects.
        \item Estimates use WBES weights and robust standard errors, not clustered standard errors.
        \item $\beta_c$ is the non-exporter slope; $\beta_c + \delta_c$ is the exporter slope.
        \item The focal estimate is $\delta_c$, the exporter minus non-exporter slope difference.
        \item Display rule: at least 30 firms overall and at least 10 firms in each exporter group.
    \end{itemize}
\end{frame}

\begin{frame}{Cameroon WBES Division Specification}

\begin{frame}{Supporting WBES Export-Value Specification}

        \item $Z_i$ controls for log firm age, foreign ownership share, and government/state ownership share.
        \item $\lambda_{q(i)}$ are ISIC Rev.4 section fixed effects.
        \item Rows with fewer than 30 observations are suppressed; benchmark rows are equal-country averages.

\begin{frame}[label=app-ecofin-classification]{Ecofin Sector Classification Notes}
    \small
    \begin{itemize}
        \item ISIC Rev.4 sections are based on the NACAM--ISIC Rev.4 crosswalk. Asterisks identify sections containing branches that span section boundaries, including combined sections.
        \item Combined sections: professional \& admin. support; arts/recreation, other services \& household employers. Household employers cover division 97 only.
        \item Mining includes branches 005 and 006, whose definitions also span manufacturing and, for 005, transport.
        \item Agriculture includes branch 009 (cocoa, coffee, tea and sugar, including processing); manufacturing includes branch 021 (rubber and plastics, with agriculture references).
        \item Electricity includes branch 029, which also spans manufacturing; real estate includes branch 037, which also spans construction.
        \item These branch assignments retain the reviewed crosswalk decisions. Sector coverage may differ across Census, Ecofin and WBES because of the source classifications.
    \end{itemize}
